# Transitive, pinned deliberately.
#
# DVC 3.x (via scmrepo) imports the private constant `_DIR_MARK` from
# pathspec.patterns.gitwildmatch. pathspec 0.12 removed it, so an unconstrained
# install resolves to 0.12.x and every dvc command dies with:
#
#   ImportError: cannot import name '_DIR_MARK' from
#   'pathspec.patterns.gitwildmatch'
#
# DVC's own metadata does not cap this, so the cap has to live here.
pathspec==0.11.2
